-- Helper: create Instance from properties table.
local function make(class, props)
	local inst = Instance.new(class)
	for k, v in pairs(props) do
		if k ~= "Parent" then inst[k] = v end
	end
	if props.Parent then inst.Parent = props.Parent end
	return inst
end

local function corner(parent, r)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, r or 8)
	c.Parent = parent
	return c
end

local function pad(parent, t, r, b, l)
	local p = Instance.new("UIPadding")
	p.PaddingTop    = UDim.new(0, t or 8)
	p.PaddingRight  = UDim.new(0, r or 8)
	p.PaddingBottom = UDim.new(0, b or 8)
	p.PaddingLeft   = UDim.new(0, l or 8)
	p.Parent = parent
	return p
end

local function listLayout(parent, dir, gap)
	return make("UIListLayout", {
		FillDirection = dir or Enum.FillDirection.Vertical,
		SortOrder = Enum.SortOrder.LayoutOrder,
		Padding = UDim.new(0, gap or 6),
		Parent = parent,
	})
end

local function uiStroke(parent, color, thickness)
	return make("UIStroke", {
		Color     = color or C_BORDER,
		Thickness = thickness or 1,
		Parent    = parent,
	})
end

local function tw(inst, props, dur, style, dir)
	tweenService:Create(
		inst,
		TweenInfo.new(dur or 0.2, style or Enum.EasingStyle.Quart, dir or Enum.EasingDirection.Out),
		props
	):Play()
end

---Send a notification toast to the top-right corner.
---@param options table
function Library:Notify(options)
	options = options or {}
	local title    = options.Title    or "Notification"
	local content  = options.Content  or ""
	local duration = options.Duration or 4
	local nType    = options.Type     or "Info"

	local accentMap = {
		Info    = C_ACCENT2,
		Success = C_SUCCESS,
		Warning = C_WARN,
		Error   = C_RED,
	}
	local accentColor = accentMap[nType] or C_ACCENT2

	local notif = make("Frame", {
		Size               = UDim2.new(1, 0, 0, 0),
		AutomaticSize      = Enum.AutomaticSize.Y,
		BackgroundColor3   = C_SURFACE,
		BorderSizePixel    = 0,
		ClipsDescendants   = true,
		BackgroundTransparency = 1,
		Parent             = notifHolder,
	})
	corner(notif, 8)
	uiStroke(notif, C_BORDER)

	-- Left accent stripe.
	local stripe = make("Frame", {
		Size             = UDim2.new(0, 3, 1, 0),
		BackgroundColor3 = accentColor,
		BorderSizePixel  = 0,
		Parent           = notif,
	})
	corner(stripe, 2)

	local inner = make("Frame", {
		Size                   = UDim2.new(1, -14, 0, 0),
		Position               = UDim2.new(0, 14, 0, 0),
		AutomaticSize          = Enum.AutomaticSize.Y,
		BackgroundTransparency = 1,
		Parent                 = notif,
	})
	pad(inner, 10, 10, 10, 6)
	listLayout(inner, Enum.FillDirection.Vertical, 3)

	make("TextLabel", {
		Size             = UDim2.new(1, 0, 0, 16),
		BackgroundTransparency = 1,
		Text             = title,
		TextColor3       = C_TEXT,
		TextSize         = 13,
		Font             = Enum.Font.GothamBold,
		TextXAlignment   = Enum.TextXAlignment.Left,
		Parent           = inner,
	})
	make("TextLabel", {
		Size             = UDim2.new(1, 0, 0, 0),
		AutomaticSize    = Enum.AutomaticSize.Y,
		BackgroundTransparency = 1,
		Text             = content,
		TextColor3       = C_MUTED,
		TextSize         = 11,
		Font             = Enum.Font.Gotham,
		TextXAlignment   = Enum.TextXAlignment.Left,
		TextWrapped      = true,
		Parent           = inner,
	})

	-- Slide in.
	tw(notif, { BackgroundTransparency = 0 }, 0.2)

	task.delay(duration, function()
		tw(notif, { BackgroundTransparency = 1 }, 0.3)
		task.delay(0.35, function() notif:Destroy() end)
	end)
end

		return btn

	-- Activate a tab by name.
	local function activateTab(name)
		if currentTab == name then return end
		-- Deactivate all.
		for _, b in ipairs(tabBtns) do
			tw(b, { BackgroundColor3 = C_SIDEBAR, BackgroundTransparency = 1 })
			local lbl = b:FindFirstChildWhichIsA("TextLabel")
			if lbl then tw(lbl, { TextColor3 = C_MUTED }) end
			local img = b:FindFirstChildWhichIsA("ImageLabel")
			if img then tw(img, { ImageColor3 = C_MUTED }) end
			local dot = b:FindFirstChild("ActiveDot")
			if dot then dot.Visible = false end
		end
		for _, f in pairs(tabFrames) do f.Visible = false end
		-- Activate target.
		local btn = tabMap[name]
		if btn then
			tw(btn, { BackgroundColor3 = C_ELEMENT, BackgroundTransparency = 0 })
			local lbl = btn:FindFirstChildWhichIsA("TextLabel")
			if lbl then tw(lbl, { TextColor3 = C_TEXT }) end
			local img = btn:FindFirstChildWhichIsA("ImageLabel")
			if img then tw(img, { ImageColor3 = C_ACCENT2 }) end
			local dot = btn:FindFirstChild("ActiveDot")
			if dot then dot.Visible = true end
		end
		if tabFrames[name] then tabFrames[name].Visible = true end
		topTabLabel.Text = name
		currentTab       = name
	end

	---Create a new sidebar tab.
	---@param name string
	---@param icon string? rbxassetid
	function Window:CreateTab(name, icon)
		local isFirst = currentTab == nil

		-- Sidebar button.
		local tabBtn = make("Frame", {
			Name                   = name .. "_Btn",
			Size                   = UDim2.new(1, 0, 0, 34),
			BackgroundColor3       = isFirst and C_ELEMENT or C_SIDEBAR,
			BackgroundTransparency = isFirst and 0 or 1,
			BorderSizePixel        = 0,
			Parent                 = tabListScroll,
		})
		corner(tabBtn, 7)

		-- Left active indicator.
		local activeDot = make("Frame", {
			Name             = "ActiveDot",
			Size             = UDim2.new(0, 3, 0, 16),
			Position         = UDim2.new(0, 0, 0.5, -8),
			BackgroundColor3 = C_ACCENT2,
			BorderSizePixel  = 0,
			Visible          = isFirst,
			Parent           = tabBtn,
		})
		corner(activeDot, 2)

		local iconOffset = 12

		if icon then
			iconOffset = 32
			make("ImageLabel", {
				Size             = UDim2.new(0, 16, 0, 16),
				Position         = UDim2.new(0, 10, 0.5, -8),
				BackgroundTransparency = 1,
				Image            = icon,
				ImageColor3      = isFirst and C_ACCENT2 or C_MUTED,
				Parent           = tabBtn,
			})
		end

		local tabLabel = make("TextLabel", {
			Size             = UDim2.new(1, -iconOffset, 1, 0),
			Position         = UDim2.new(0, iconOffset, 0, 0),
			BackgroundTransparency = 1,
			Text             = name,
			TextColor3       = isFirst and C_TEXT or C_MUTED,
			TextSize         = 13,
			Font             = Enum.Font.GothamSemibold,
			TextXAlignment   = Enum.TextXAlignment.Left,
			Parent           = tabBtn,
		})

		-- Tab content frame inside shared content scroll.
		local tabFrame = make("Frame", {
			Name                   = name .. "_Frame",
			Size                   = UDim2.new(1, 0, 1, 0),
			BackgroundTransparency = 1,
			Visible                = isFirst,
			Parent                 = contentScroll,
		})
		pad(tabFrame, 10, 12, 10, 12)
		listLayout(tabFrame, Enum.FillDirection.Vertical, 8)

		table.insert(tabBtns, tabBtn)
		tabMap[name]    = tabBtn
		tabFrames[name] = tabFrame

		if isFirst then
			currentTab         = name
			topTabLabel.Text   = name
		end

		-- Click and hover events.
		tabBtn.InputBegan:Connect(function(i)
			if i.UserInputType == Enum.UserInputType.MouseButton1 then activateTab(name) end
		end)
		tabBtn.MouseEnter:Connect(function()
			if currentTab ~= name then
				tw(tabBtn, { BackgroundColor3 = C_HOVER, BackgroundTransparency = 0 })
				tw(tabLabel, { TextColor3 = Color3.fromRGB(150, 150, 175) })
			end
		end)
		tabBtn.MouseLeave:Connect(function()
			if currentTab ~= name then
				tw(tabBtn, { BackgroundTransparency = 1 })
				tw(tabLabel, { TextColor3 = C_MUTED })
			end
		end)

			local Section = {}

			---Create a toggle.
			---@param options table
			function Section:CreateToggle(options)
				options   = options or {}
				local lbl = options.Name        or "Toggle"
				local dsc = options.Description or ""
				local val = options.CurrentValue or false
				local cb  = options.Callback     or function() end

				Toggle.CurrentValue = state
				function Toggle:Set(v) set(v); self.CurrentValue = v end
				function Toggle:Get() return state end

			end

			---Create a slider.
			---@param options table
			function Section:CreateSlider(options)

				local lbl = options.Name         or "Slider"

				local dsc = options.Description  or ""
				local cb  = options.Callback     or function() end

				local minV, maxV = rng[1], rng[2]
				local value      = math.clamp(cur, minV, maxV)
				local rowH       = dsc ~= "" and 62 or 52

				local row = make("Frame", {
					Size             = UDim2.new(1, 0, 0, rowH),

					Size             = UDim2.new(1, -68, 0, 16),
					Position         = UDim2.new(0, 10, 0, 7),
					BackgroundTransparency = 1,
					Text             = lbl,
					TextColor3       = C_TEXT,
					TextSize         = 13,
					Font             = Enum.Font.GothamSemibold,
					TextXAlignment   = Enum.TextXAlignment.Left,
					Parent           = row,
				})

				local valLbl = make("TextLabel", {
					Size             = UDim2.new(0, 58, 0, 16),
					Position         = UDim2.new(1, -68, 0, 7),
					BackgroundTransparency = 1,
					Text             = tostring(value) .. sfx,
					TextColor3       = C_ACCENT2,
					TextSize         = 12,
					Font             = Enum.Font.GothamBold,
					TextXAlignment   = Enum.TextXAlignment.Right,
					Parent           = row,
				})

				if dsc ~= "" then
					make("TextLabel", {
						Size             = UDim2.new(1, -68, 0, 13),
						Position         = UDim2.new(0, 10, 0, 24),
						BackgroundTransparency = 1,
						Text             = dsc,
						TextColor3       = C_MUTED,
						TextSize         = 11,
						Font             = Enum.Font.Gotham,
						TextXAlignment   = Enum.TextXAlignment.Left,
						Parent           = row,
					})
				end

				local trackY = dsc ~= "" and 46 or 36

				local sliderTrack = make("Frame", {
					Size             = UDim2.new(1, -20, 0, 4),
					Position         = UDim2.new(0, 10, 0, trackY),
					BackgroundColor3 = C_TRACK,
					BorderSizePixel  = 0,
					Parent           = row,
				})
				corner(sliderTrack, 2)

				local rel0 = (value - minV) / (maxV - minV)

				local fill = make("Frame", {
					Size             = UDim2.new(rel0, 0, 1, 0),
					BackgroundColor3 = C_ACCENT,
					BorderSizePixel  = 0,
					Parent           = sliderTrack,
				})
				corner(fill, 2)

				local nub = make("Frame", {
					Size             = UDim2.new(0, 12, 0, 12),
					Position         = UDim2.new(rel0, -6, 0.5, -6),
					BackgroundColor3 = C_WHITE,
					BorderSizePixel  = 0,
					Parent           = sliderTrack,
				})
				corner(nub, 6)

				local sliding = false

				local function updateSlider(x)
					local rel = math.clamp((x - sliderTrack.AbsolutePosition.X) / sliderTrack.AbsoluteSize.X, 0, 1)
					value = math.round((minV + (maxV - minV) * rel) / inc) * inc
					value = math.clamp(value, minV, maxV)
					local r = (value - minV) / (maxV - minV)
					valLbl.Text  = tostring(value) .. sfx
					fill.Size    = UDim2.new(r, 0, 1, 0)
					nub.Position = UDim2.new(r, -6, 0.5, -6)
					if cfgEnabled then configData[lbl] = value end
					cb(value)
				end

				sliderTrack.InputBegan:Connect(function(i)
					if i.UserInputType == Enum.UserInputType.MouseButton1 then sliding = true; updateSlider(i.Position.X) end
				end)
				userInputService.InputChanged:Connect(function(i)
					if sliding and i.UserInputType == Enum.UserInputType.MouseMovement then updateSlider(i.Position.X) end
				end)
				userInputService.InputEnded:Connect(function(i)
					if i.UserInputType == Enum.UserInputType.MouseButton1 then sliding = false end
				end)

				local Slider = {}
				Slider.CurrentValue = value
				function Slider:Set(v)
					value = math.clamp(v, minV, maxV)
					local r = (value - minV) / (maxV - minV)
					valLbl.Text  = tostring(value) .. sfx
					fill.Size    = UDim2.new(r, 0, 1, 0)
					nub.Position = UDim2.new(r, -6, 0.5, -6)
					self.CurrentValue = value
				end
				function Slider:Get() return value end
				return Slider
			end

				local cur  = options.CurrentOption   or (opts[1] or "None")
				local dsc  = options.Description     or ""

					Font             = Enum.Font.GothamBold,

				})

					Size             = UDim2.new(1, -20, 0, 0),

				listLayout(listScroll, Enum.FillDirection.Vertical, 2)

				local open          = false
				local maxListHeight = math.min(#opts * 28 + 8, 128)

				local function buildList()
					for _, c in ipairs(listScroll:GetChildren()) do
						if c:IsA("TextButton") then c:Destroy() end
					end
					for _, item in ipairs(opts) do
						local itemBtn = make("TextButton", {
							Size             = UDim2.new(1, 0, 0, 26),
							BackgroundColor3 = C_SURFACE,
							Text             = tostring(item),
							TextColor3       = C_TEXT,

							ZIndex           = 13,
							Parent           = listScroll,

						itemBtn.MouseEnter:Connect(function() tw(itemBtn, { BackgroundColor3 = C_HOVER }) end)
						itemBtn.MouseLeave:Connect(function()
							local isSelected = multi and table.find(selected, item) or selected == item
							tw(itemBtn, { BackgroundColor3 = isSelected and C_ACCENT or C_SURFACE })
						end)

						itemBtn.MouseButton1Click:Connect(function()
							if not multi then
								selected   = item
								selLbl.Text = tostring(item)
								open = false
								tw(listPopup, { Size = UDim2.new(1, -20, 0, 0) }, 0.14)
								tw(chevron, { Rotation = 0 })
								task.delay(0.15, function() listPopup.Visible = false end)
								cb(selected)
							else
								local idx = table.find(selected, item)
								if idx then
									table.remove(selected, idx)
									tw(itemBtn, { BackgroundColor3 = C_SURFACE })
								else
									table.insert(selected, item)
									tw(itemBtn, { BackgroundColor3 = C_ACCENT })
								end
								selLbl.Text = #selected > 0 and table.concat(selected, ", ") or "Select..."
								cb(selected)
							end
						end)
					end
				end
				buildList()

				dropBtn.MouseButton1Click:Connect(function()
					open = not open
					if open then
						listPopup.Visible = true
						tw(listPopup, { Size = UDim2.new(1, -20, 0, maxListHeight) }, 0.2)
						tw(chevron, { Rotation = 180 })
					else
						tw(listPopup, { Size = UDim2.new(1, -20, 0, 0) }, 0.14)
						tw(chevron, { Rotation = 0 })
						task.delay(0.15, function() listPopup.Visible = false end)
					end
				end)

				local Dropdown = {}
				Dropdown.CurrentOption = selected
				function Dropdown:Set(v)
					selected   = v
					selLbl.Text = tostring(v)
					self.CurrentOption = v
				end
				function Dropdown:Refresh(newOpts)
					opts = newOpts
					maxListHeight = math.min(#opts * 28 + 8, 128)
					buildList()
				end
				function Dropdown:Get() return selected end
				return Dropdown
			end

	---Save config to file.
	function Window:SaveConfig()
		if not cfgEnabled then return end
		if not isfolder(cfgFolder) then makefolder(cfgFolder) end
		writefile(cfgFolder .. "/" .. cfgFile .. ".json", httpService:JSONEncode(configData))
	end

	---Load config from file.
	function Window:LoadConfig()
		if not cfgEnabled then return end
		local path = cfgFolder .. "/" .. cfgFile .. ".json"
		if not isfile(path) then return end
		local ok, data = pcall(httpService.JSONDecode, httpService, readfile(path))
		if ok and type(data) == "table" then configData = data end
	end

	---Destroy the window.
	function Window:Destroy()
		tw(win, { Size = UDim2.new(0, WINDOW_W, 0, 0), BackgroundTransparency = 1 }, 0.22)
		task.delay(0.25, function() gui:Destroy() end)
	end

	return Window
end

function Library:Destroy()
	gui:Destroy()
end
